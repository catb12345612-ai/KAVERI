Step 3: Google Apps Script (Code.gs) — Auto Daily Google Sheet Sync 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  var sheet = ss.getSheetByName(payload.type) || ss.insertSheet(payload.type);
  sheet.appendRow([new Date(), JSON.stringify(payload.data)]);
  return ContentService.createTextOutput(JSON.stringify({status: "SYNCED"}))
    .setMimeType(ContentService.MimeType.JSON);
}
